cmake_minimum_required(VERSION 3.22)
project(lumaribbon VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
option(LUMA_ANALYSIS_ONLY "Build only the Qt-independent analysis library and tests" OFF)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3f)

add_library(luma_analysis STATIC src/SignalAnalyzer.cpp src/AudioProcessor.cpp src/RibbonMotion.cpp)
target_include_directories(luma_analysis PUBLIC src)
target_link_libraries(luma_analysis PUBLIC PkgConfig::FFTW)
target_compile_options(luma_analysis PRIVATE -Wall -Wextra -Wpedantic)

if(BUILD_TESTING)
    foreach(suite IN ITEMS analysis musical-response color-response motion)
        string(REPLACE "-" "_" source_name "${suite}")
        add_executable(test-${suite} tests/test_${source_name}.cpp)
        target_link_libraries(test-${suite} PRIVATE luma_analysis Threads::Threads)
        add_test(NAME ${suite} COMMAND test-${suite})
    endforeach()
endif()

if(LUMA_ANALYSIS_ONLY)
    return()
endif()

set(CMAKE_AUTOMOC ON)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Qml Quick ShaderTools)
find_package(KF6Config 6.0 REQUIRED)
find_package(KF6CoreAddons 6.0 REQUIRED)
find_package(Plasma 6.0 REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.65)

add_library(luma_audio STATIC src/PipeWireCapture.cpp src/AudioEngine.cpp src/AudioState.cpp)
target_link_libraries(luma_audio PUBLIC luma_analysis Qt6::Core Threads::Threads PRIVATE PkgConfig::PIPEWIRE)
target_compile_options(luma_audio PRIVATE -Wall -Wextra -Wpedantic)

add_library(lumaribbon MODULE src/LumaApplet.cpp)
target_link_libraries(lumaribbon PRIVATE luma_audio Plasma::Plasma KF6::CoreAddons Qt6::Quick)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS package/metadata.json)
file(READ package/metadata.json native_metadata)
string(JSON native_metadata REMOVE "${native_metadata}" KPlugin Id)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-metadata.json" CONTENT "${native_metadata}\n")
target_include_directories(lumaribbon PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(lumaribbon PROPERTIES PREFIX "" OUTPUT_NAME "org.kde.plasma.lumaribbon"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/plasma/applets")
qt_add_shaders(lumaribbon ribbon_shaders PREFIX "/lumaribbon" BASE shaders FILES shaders/ribbon.frag)

set(LUMA_PLUGIN_INSTALL_DIR "${KDE_INSTALL_PLUGINDIR}/plasma/applets" CACHE PATH "Plasma applet plugin directory")
install(TARGETS lumaribbon DESTINATION "${LUMA_PLUGIN_INSTALL_DIR}")
install(DIRECTORY package/ DESTINATION "${CMAKE_INSTALL_DATADIR}/plasma/plasmoids/org.kde.plasma.lumaribbon")
install(FILES icons/hicolor/scalable/apps/org.kde.plasma.lumaribbon.svg
    DESTINATION "${KDE_INSTALL_ICONDIR}/hicolor/scalable/apps")
install(FILES README.md LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/lumaribbon")
install(DIRECTORY docs/ DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/lumaribbon/docs")
configure_file(cmake/uninstall.cmake.in uninstall.cmake @ONLY)
add_custom_target(uninstall COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_BINARY_DIR}/uninstall.cmake")

option(LUMA_BUILD_PREVIEW "Build the standalone Qt Quick audio/visual verification tool" ON)
if(LUMA_BUILD_PREVIEW)
    add_executable(luma-preview tools/preview.cpp)
    target_link_libraries(luma-preview PRIVATE luma_audio Qt6::Quick)
    target_compile_definitions(luma-preview PRIVATE LUMA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    qt_add_shaders(luma-preview preview_shaders PREFIX "/lumaribbon" BASE shaders FILES shaders/ribbon.frag)
endif()

option(LUMA_BUILD_MOTION_PROTOTYPE "Build the disposable main-shape comparison tool" OFF)
if(LUMA_BUILD_MOTION_PROTOTYPE)
    add_subdirectory(tools/macro-motion-prototype)
endif()

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    find_package(KF6Package 6.0 REQUIRED)
    find_package(PlasmaQuick 6.0 REQUIRED)
    add_executable(luma-audio-probe tools/audio_probe.cpp)
    target_link_libraries(luma-audio-probe PRIVATE luma_audio)
    add_executable(test-engine-recovery tests/test_engine_recovery.cpp tests/fftw_failure.cpp)
    target_link_libraries(test-engine-recovery PRIVATE luma_audio ${CMAKE_DL_LIBS})
    add_test(NAME engine-recovery COMMAND test-engine-recovery)
    add_test(NAME engine-unexpected-recovery COMMAND test-engine-recovery --unexpected)
    set_tests_properties(engine-recovery engine-unexpected-recovery PROPERTIES TIMEOUT 15)
    # Deterministic capture adapter exercises the actual engine/state lifecycle
    # and GUI wake-up queue without using the desktop's audio server.
    add_executable(test-audio-state tests/test_audio_state.cpp src/AudioEngine.cpp src/AudioState.cpp)
    target_link_libraries(test-audio-state PRIVATE luma_analysis Qt6::Test Threads::Threads)
    add_test(NAME audio-state COMMAND test-audio-state)
    set_tests_properties(audio-state PROPERTIES TIMEOUT 30)
    add_executable(test-capture tests/test_capture.cpp)
    target_link_libraries(test-capture PRIVATE luma_audio)
    add_test(NAME capture COMMAND test-capture)
    set_tests_properties(capture PROPERTIES TIMEOUT 10)
    add_executable(luma-test-audio tests/write_test_audio.cpp)
    add_executable(test-view tests/test_view.cpp)
    target_link_libraries(test-view PRIVATE Qt6::Quick Qt6::Test)
    target_compile_definitions(test-view PRIVATE LUMA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    qt_add_shaders(test-view test_shaders PREFIX "/lumaribbon" BASE shaders FILES shaders/ribbon.frag)
    add_test(NAME view COMMAND test-view)
    add_executable(test-motion-view tests/test_motion_view.cpp)
    target_link_libraries(test-motion-view PRIVATE luma_analysis Qt6::Quick Qt6::Test)
    target_compile_definitions(test-motion-view PRIVATE LUMA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    qt_add_shaders(test-motion-view motion_test_shaders PREFIX "/lumaribbon" BASE shaders FILES shaders/ribbon.frag)
    add_test(NAME motion-view COMMAND test-motion-view)
    set_tests_properties(motion-view PROPERTIES TIMEOUT 60)
    add_executable(test-plasma tests/test_plasma.cpp)
    target_link_libraries(test-plasma PRIVATE Plasma::Plasma Plasma::PlasmaQuick KF6::Package KF6::ConfigQml Qt6::Quick Qt6::Test)
    target_compile_definitions(test-plasma PRIVATE LUMA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_custom_target(stage-luma-test-package
        COMMAND "${CMAKE_COMMAND}" -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/package"
            "${CMAKE_BINARY_DIR}/test-data/plasma/plasmoids/org.kde.plasma.lumaribbon"
        COMMAND "${CMAKE_COMMAND}" -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/icons"
            "${CMAKE_BINARY_DIR}/test-data/icons"
        COMMENT "Refreshing the Plasma test package")
    add_dependencies(test-plasma lumaribbon stage-luma-test-package)
    # Uses the staged package and the freshly built plugin, never the user's panel.
    add_test(NAME plasma COMMAND test-plasma)
    set_tests_properties(plasma PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins;XDG_DATA_DIRS=${CMAKE_BINARY_DIR}/test-data:/usr/local/share:/usr/share;XDG_CONFIG_HOME=${CMAKE_BINARY_DIR}/test-config;LANGUAGE=en_US")
endif()
